# Q2 and Q72

## Q2: the Gaussian unit

Q2 is two bits: the sign and the imaginary bit, read together as the quarter-turn count, so the four units form the group $\mathbb{Z}/4$ and multiplication is addition of turns. A byte is the smallest unit a machine addresses, so a Q2 stored alone takes a byte, in its top two bits. Inside a Q72 the same byte's low six bits are the slash; alone they mean nothing, so the canonical byte of a Q2 has them clear.

```python {file=build/q2.py}
UNIT_NAMES = ("1", "i", "-1", "-i")
SIGN_BIT, IMAG_BIT = 0x80, 0x40
SLASH_MASK = 0x3F


def turn_of(code):
    """The quarter-turn count of a Q2 byte: (sign bit, imaginary bit) read as a 2-bit number,
    so 0 = 1, 1 = i, 2 = -1, 3 = -i."""
    return code >> 6


class Q2:
    """Q2, the Gaussian unit i**t: two bits, the sign and the imaginary bit. Stored alone it occupies a byte,
    in the top two bits; inside a Q72 the same byte's low six bits are the slash. Alone, those six bits mean
    nothing, so the canonical byte of a Q2 has them clear."""

    __slots__ = ("turn",)

    def __init__(self, turn):
        self.turn = turn % 4

    @staticmethod
    def from_byte(code):
        if not 0 <= code <= 255:
            raise ValueError("a Q2 is stored in one byte")
        return Q2(turn_of(code))

    def byte(self, slash=0):
        return (self.turn << 6) | slash

    @property
    def negative(self):
        return self.turn >= 2

    @property
    def imaginary(self):
        return self.turn & 1 == 1

    def __mul__(self, other):
        return Q2(self.turn + other.turn)

    def reciprocal(self):
        return Q2(-self.turn)

    def __truediv__(self, other):
        return self * other.reciprocal()

    def __neg__(self):
        return Q2(self.turn + 2)

    def conj(self):
        return Q2(-self.turn)

    def __eq__(self, other):
        return isinstance(other, Q2) and self.turn == other.turn

    def __hash__(self):
        return hash(("Q2", self.turn))

    def __repr__(self):
        return f"Q2({UNIT_NAMES[self.turn]})"
```

## Q72: the encoding

For a slash $s$ and a U64 $u$, the numerator is the top $s+1$ bits and the remaining $63-s$ bits hold $q-1$:

$$
p = \left\lfloor u / 2^{63-s} \right\rfloor, \qquad q = (u \bmod 2^{63-s}) + 1, \qquad x = i^t\,\frac{p}{q}.
$$

Every pattern decodes. A value $p/q$ in lowest terms fits when $\operatorname{bits}(p) + \operatorname{bits}(q-1) \le 64$, and its canonical encoding uses $s = \operatorname{bits}(p) - 1$. Zero is any pattern whose numerator is 0; canonically all bits clear. `fits`, `encode` and `decode` take the word size as a parameter, so the same code describes the whole ladder, and the tests check it exhaustively on a 16-bit word.

```python {file=build/q72.py}
import math
from fractions import Fraction

from q2 import Q2

WORD = 64
U64_MAX = 2**64 - 1


class CrossAxis(ArithmeticError):
    """A sum of a nonzero real and a nonzero imaginary value: a complex number, not one Q72."""


def bitlen(x):
    return x.bit_length()


def fits(p, q, word=WORD):
    """Does i**t * p/q (p >= 0, q >= 1, in lowest terms) fit one pair of a G and a `word`-bit unsigned integer?
    The numerator takes the top bits(p) bits, so bits(p) - 1 is the slash, and the denominator is stored
    minus one in the remaining word - bits(p) bits."""
    if p == 0:
        return True
    bp = p.bit_length()
    return bp <= word and (q - 1).bit_length() <= word - bp


def encode(turn, p, q, word=WORD):
    """The canonical (G byte, unsigned) of i**turn * p/q. Zero is (0, 0)."""
    if p == 0:
        return 0, 0
    if p < 0 or q < 1 or math.gcd(p, q) != 1:
        raise ValueError("encode takes a nonnegative numerator and positive denominator in lowest terms")
    if not fits(p, q, word):
        raise OverflowError("does not fit one pair")
    bp = p.bit_length()
    return ((turn % 4) << 6) | (bp - 1), (p << (word - bp)) | (q - 1)


def decode(code, u, word=WORD):
    """(turn, p, q) of any pair: every bit pattern is a value. The slash s puts the numerator in the top s+1
    bits and the stored denominator (q - 1) in the rest; slash word-1 leaves no denominator field, so q = 1.
    A numerator of 0 is zero, whatever the G byte holds; zero is reported with turn 0."""
    if not (0 <= code <= 255 and 0 <= u < 2**word):
        raise ValueError("a pair is one byte and one unsigned word")
    s = code & (word - 1)
    width = word - 1 - s
    p, q = u >> width, (u & ((1 << width) - 1)) + 1
    if p == 0:
        return 0, 0, 1
    return code >> 6, p, q


def is_canonical(code, u, word=WORD):
    turn, p, q = decode(code, u, word)
    if p == 0:
        return code == 0 and u == 0
    return (code & (word - 1)) == p.bit_length() - 1 and math.gcd(p, q) == 1


```

## Exact values: `Q`

`Q` is the exact reference value: a turn and a nonnegative rational magnitude, unbounded, with zero always at turn 0. Products add turns; sums are defined along one axis, and a real plus an imaginary raises `CrossAxis`, because that is a complex number of two values. Division by zero raises `ZeroDivisionError`. Real values are ordered; imaginary ones are not.

```python {file=build/q72.py}
class Q:
    """An exact axis-aligned value i**turn * mag (mag a nonnegative Fraction; zero has turn 0).

    It is not limited to one pair: fits() and word() say whether and how it fits. A Q that does not fit is what
    the prototype's promotion stand-in stores."""

    __slots__ = ("turn", "mag")

    def __init__(self, turn, mag):
        mag = Fraction(mag)
        if mag < 0:
            raise ValueError("the magnitude is nonnegative; the direction is the turn")
        self.mag = mag
        self.turn = 0 if mag == 0 else turn % 4

    @staticmethod
    def of(x, turn=0):
        x = Fraction(x)
        return Q(turn + (2 if x < 0 else 0), abs(x))

    @staticmethod
    def imag(x):
        return Q.of(x, 1)

    @staticmethod
    def from_word(code, u):
        turn, p, q = decode(code, u)
        return Q(turn, Fraction(p, q))

    @property
    def unit(self):
        return Q2(self.turn)

    @property
    def axis(self):
        return self.turn & 1

    def is_zero(self):
        return self.mag == 0

    def fits(self, word=WORD):
        return fits(self.mag.numerator, self.mag.denominator, word)

    def word(self):
        if not self.fits():
            return None
        return encode(self.turn, self.mag.numerator, self.mag.denominator)

    def parts(self):
        v = self.mag if self.turn < 2 else -self.mag
        return (v, Fraction(0)) if self.axis == 0 else (Fraction(0), v)

    def fraction(self):
        re, im = self.parts()
        if im:
            raise ValueError("imaginary")
        return re

    def __neg__(self):
        return Q(self.turn + 2, self.mag)

    def conj(self):
        return Q(-self.turn, self.mag)

    def reciprocal(self):
        if self.mag == 0:
            raise ZeroDivisionError("division by zero")
        return Q(-self.turn, 1 / self.mag)

    def __mul__(self, o):
        o = coerce(o)
        return Q(self.turn + o.turn, self.mag * o.mag)

    __rmul__ = __mul__

    def __truediv__(self, o):
        return self * coerce(o).reciprocal()

    def __rtruediv__(self, o):
        return coerce(o) / self

    def __add__(self, o):
        o = coerce(o)
        if self.mag == 0:
            return o
        if o.mag == 0:
            return self
        if self.axis != o.axis:
            raise CrossAxis("real plus imaginary")
        va = self.mag if self.turn < 2 else -self.mag
        vb = o.mag if o.turn < 2 else -o.mag
        s = va + vb
        return Q(self.axis + (2 if s < 0 else 0), abs(s))

    __radd__ = __add__

    def __sub__(self, o):
        return self + (-coerce(o))

    def __rsub__(self, o):
        return coerce(o) - self

    def _real(self):
        if self.mag != 0 and self.axis:
            raise TypeError("imaginary values are not ordered")
        return self.mag if self.turn == 0 else -self.mag

    def __lt__(self, o):
        return self._real() < coerce(o)._real()

    def __le__(self, o):
        return self._real() <= coerce(o)._real()

    def __gt__(self, o):
        return self._real() > coerce(o)._real()

    def __ge__(self, o):
        return self._real() >= coerce(o)._real()

    def __eq__(self, o):
        if not isinstance(o, (Q, int, Fraction)):
            return NotImplemented
        o = coerce(o)
        return self.turn == o.turn and self.mag == o.mag

    def __hash__(self):
        if self.turn in (0, 2):
            return hash(self.mag if self.turn == 0 else -self.mag)
        return hash(("Q", self.turn, self.mag))

    def __repr__(self):
        return f"Q({Q2(self.turn)!r}*{self.mag})" if self.mag else "Q(0)"


def coerce(x):
    if isinstance(x, Q):
        return x
    if isinstance(x, (int, Fraction)):
        return Q.of(x)
    raise TypeError(f"cannot use {type(x).__name__} as an exact value")
```
